function sendWhatsAppOrder() { const nameInput = document.getElementById('cust-name'); const name = nameInput ? nameInput.value.trim() : ''; if (!name) { alert("Inserisci il tuo nome prima di procedere!"); if(nameInput) nameInput.focus(); return; } const addrInput = document.getElementById('cust-addr'); const addr = addrInput ? addrInput.value.trim() : ''; const timeInput = document.getElementById('cust-time'); const time = timeInput ? timeInput.value.trim() : ''; const hidePrices = (state.orderType === 'deliveroo'); let orderText = `*Ordine, ${getOrderCode()}*\n`; orderText += `*Kebab di Via Roma*\n`; orderText += `*Tipo:* ${state.orderType.toUpperCase()}\n`; orderText += `*Nome:* ${name}\n`; if (addr) orderText += `*Indirizzo:* ${addr}\n`; if (time) orderText += `*Orario:* ${time}\n`; orderText += `\n--------------------------\n*DETTAGLIO CUCINA:*\n`; let foodSubtotalRaw = 0; let foodSubtotalDiscounted = 0; let totalAsportoDiscount = 0; let itemCounter = 1; for (let id in state.cart) { const item = menuData.find(m => m.id === id); const cartItem = state.cart[id]; let qty = cartItem.qty; let localizedTitle = getLocalizedName(item.name); if (item.isBeverage) { let effPrice = item.price; foodSubtotalRaw += effPrice * qty; foodSubtotalDiscounted += effPrice * qty; orderText += `\n*${itemCounter}.* ${qty}x ${localizedTitle}`; if (!hidePrices) { orderText += ` (€ ${(effPrice * qty).toFixed(2)})`; } orderText += `\n`; itemCounter++; } else if (item.isAlcohol) { let rawPrice = item.price; let effPrice = item.price; let isDiscountable = (state.orderType === 'asporto'); if (isDiscountable) { effPrice = Math.max(0, effPrice - 0.50); totalAsportoDiscount += 0.50 * qty; } foodSubtotalRaw += rawPrice * qty; foodSubtotalDiscounted += effPrice * qty; orderText += `\n*${itemCounter}.* ${qty}x ${localizedTitle}`; if (!hidePrices) { orderText += ` (€ ${(effPrice * qty).toFixed(2)})`; } orderText += `\n`; itemCounter++; } else { orderText += `\n*${itemCounter}.* ${qty}x ${localizedTitle}\n`; cartItem.units.forEach((unit, uIdx) => { let rawPrice = item.price; let effPrice = item.price; let isDiscountable = (state.orderType === 'asporto' && !item.isStudentExclusive); if (isDiscountable) { effPrice = Math.max(0, effPrice - 0.50); totalAsportoDiscount += 0.50; } let unitExtraCost = (unit.extra && unit.extra.includes('carne_extra')) ? 0.50 : 0; foodSubtotalRaw += rawPrice + unitExtraCost; foodSubtotalDiscounted += effPrice + unitExtraCost; orderText += ` ▪ #${uIdx + 1}`; let hasSenza = unit.senza && unit.senza.length > 0; let hasExtra = unit.extra && unit.extra.length > 0; if (!hasSenza && !hasExtra && !unit.note) { orderText += ` ❤️ Completo`; } else { if (hasSenza) { let senzaNames = unit.senza.map(k => getLocalizedIngredient(k)).join(', '); orderText += ` | ❌ Senza: ${senzaNames}`; } if (hasExtra) { let extraNames = unit.extra.map(k => getLocalizedIngredient(k)).join(', '); orderText += ` | ✅ Extra: ${extraNames}`; } } if (unit.note) { orderText += ` | Note: ${unit.note}`; } orderText += `\n`; }); itemCounter++; } } let deliveryFee = (state.orderType === 'consegna') ? 2.50 : 0; let grandTotal = foodSubtotalDiscounted + deliveryFee; orderText += `\n--------------------------\n`; if (!hidePrices) { orderText += `Subtotale cibo: € ${foodSubtotalRaw.toFixed(2)}\n`; if (state.orderType === 'asporto' && totalAsportoDiscount > 0) { orderText += `Sconto Asporto: -€ ${totalAsportoDiscount.toFixed(2)}\n`; } if (deliveryFee > 0) { orderText += `Consegna (Rider): € ${deliveryFee.toFixed(2)}\n`; } if (foodSubtotalDiscounted >= 40) { orderText += `🎁 Coca-Cola 1.5L Omaggio Inclusa!\n`; } orderText += `*TOTALE COMPLESSIVO: € ${grandTotal.toFixed(2)}*\n`; } else { orderText += `*Note Deliveroo Inviate*\n`; } let encodedMessage = encodeURIComponent(orderText); let whatsappUrl = `https://wa.me/393932228224?text=${encodedMessage}`; window.open(whatsappUrl, '_blank'); state.cart = {}; localStorage.removeItem('cart'); renderMenu(); calcTotal(); closeConfirmationModal(); }
